-- CAPA 1: CATÁLOGOS MAESTROS (4 tablas)
-- Tablas de referencia que alimentan al resto del sistema
-- ═══════════════════════════════════════════════════════════════════════════════

-- ═══════════════════════════════════════════════════════════════════════════════
-- [01] IDIOMAS
-- Define los idiomas soportados: Ruso, Mandarín, y futuros escalables
-- ═══════════════════════════════════════════════════════════════════════════════
PRINT '>>> CREANDO TABLA [01] Idiomas...';
GO

CREATE TABLE Idiomas (
    IdIdioma        INT IDENTITY(1,1) PRIMARY KEY,
    Nombre          NVARCHAR(50)    NOT NULL,        -- Ej: 'Ruso', 'Mandarín'
    CodigoISO       CHAR(2)         NOT NULL,        -- Ej: 'RU', 'ZH'
    EsteticaVisual  NVARCHAR(50)    NULL,            -- CSS theme: 'soviet_style.css'

    CONSTRAINT UQ_Idiomas_CodigoISO UNIQUE (CodigoISO),
    CONSTRAINT UQ_Idiomas_Nombre UNIQUE (Nombre)
);
GO

CREATE NONCLUSTERED INDEX IX_Idiomas_Nombre ON Idiomas(Nombre);
GO

-- ═══════════════════════════════════════════════════════════════════════════════
-- [02] RANGOS
-- Sistema de progresión militar del usuario
-- ═══════════════════════════════════════════════════════════════════════════════
PRINT '>>> CREANDO TABLA [02] Rangos...';
GO

CREATE TABLE Rangos (
    IdRango         INT IDENTITY(1,1) PRIMARY KEY,
    NombreRango     NVARCHAR(50)    NOT NULL,        -- Ej: 'Recluta', 'Espía'
    NivelRequerido  INT             NOT NULL,        -- XP mínimo para alcanzar
    IconoMilitar    NVARCHAR(50)    NULL,            -- Emoji/icono del rango

    CONSTRAINT UQ_Rangos_NombreRango UNIQUE (NombreRango)
);
GO

CREATE NONCLUSTERED INDEX IX_Rangos_NivelRequerido ON Rangos(NivelRequerido);
GO

-- ═══════════════════════════════════════════════════════════════════════════════

CREATE TABLE Categorias (

-- ═══════════════════════════════════════════════════════════════════════════════
-- [04] NIVELES DE DIFICULTAD  ← NUEVA TABLA (Recomendación #2: Normalización)
-- Maestro de niveles (antes campo INT libre en Diccionario)
-- ═══════════════════════════════════════════════════════════════════════════════
PRINT '>>> CREANDO TABLA [04] NivelesDificultad...';
GO

CREATE TABLE NivelesDificultad (
    IdNivel             INT IDENTITY(1,1) PRIMARY KEY,
    NombreNivel         NVARCHAR(50)    NOT NULL,    -- 'Clasificado', 'Secreto', 'Ultra Secreto'
    Descripcion         NVARCHAR(255)   NULL,
    MultiplicadorPuntos DECIMAL(3,2)    DEFAULT 1.00, -- Bonus de dificultad
    ColorTema           NVARCHAR(20)    NULL,        -- Color UI para este nivel

    CONSTRAINT UQ_Niveles_Nombre UNIQUE (NombreNivel),
    CONSTRAINT CK_Niveles_Multiplicador CHECK (MultiplicadorPuntos > 0)
);
GO

-- ═══════════════════════════════════════════════════════════════════════════════
-- CAPA 2: NÚCLEO DE CONTENIDO (4 tablas)
-- El corazón del aprendizaje: diccionario, frases, flashcards
-- ═══════════════════════════════════════════════════════════════════════════════

-- ═══════════════════════════════════════════════════════════════════════════════
-- [05] DICCIONARIO  ← MODIFICADA (Recomendaciones #1 y #2 aplicadas)
-- Mapeo de caracteres originales, fonética/Pinyin y traducción
-- CAMBIOS: Categoria NVARCHAR → IdCategoria INT (FK)
--          NivelDificultad INT → IdNivel INT (FK)
--          NUEVO: AudioUrl para pronunciaciones (Recomendación #4)
-- ═══════════════════════════════════════════════════════════════════════════════
PRINT '>>> CREANDO TABLA [05] Diccionario...';
GO
CREATE TABLE dbo.Diccionario ( 
    IdItem              INT IDENTITY(1,1) PRIMARY KEY, 
    IdIdioma            INT             NOT NULL, 
    IdCategoria         INT             NULL,        -- ← ANTES: Categoria NVARCHAR(100) 
    IdNivel             INT             NULL,        -- ← ANTES: NivelDificultad INT 
    CaracterOriginal    NVARCHAR(255)   NOT NULL,    -- Cirílico o Hanzi 
    LecturaAyuda        NVARCHAR(255)   NULL,        -- Fonética o Pinyin 
    TraduccionEspanol   NVARCHAR(255)   NOT NULL, 
    AudioUrl            NVARCHAR(500)   NULL,        -- ← NUEVO: URL de pronunciación (Rec. #4) 
    NotasContexto       NVARCHAR(500)   NULL,        -- Uso contextual 
    FechaCreacion       DATETIME2       DEFAULT GETDATE(), 
    Activo              BIT             DEFAULT 1,

    -- Foreign Keys
    CONSTRAINT FK_Diccionario_Idioma FOREIGN KEY (IdIdioma) 
        REFERENCES dbo.Idiomas(IdIdioma) ON DELETE CASCADE,
    CONSTRAINT FK_Diccionario_Categoria FOREIGN KEY (IdCategoria) 
        REFERENCES dbo.Categorias(IdCategoria) ON DELETE SET NULL,
    CONSTRAINT FK_Diccionario_Nivel FOREIGN KEY (IdNivel) 
        REFERENCES dbo.NivelesDificultad(IdNivel) ON DELETE SET NULL,

    -- Índices estratégicos (Recomendación #3)
); 
GO

CREATE NONCLUSTERED INDEX IX_Diccionario_IdIdioma_IdCategoria ON dbo.Diccionario(IdIdioma, IdCategoria);
GO

CREATE NONCLUSTERED INDEX IX_Diccionario_IdNivel ON dbo.Diccionario(IdNivel);
GO

CREATE NONCLUSTERED INDEX IX_Diccionario_Activo_IdIdioma ON dbo.Diccionario(Activo, IdIdioma);
GO

-- Índice adicional para búsqueda por caracteres (especialmente importante para Hanzi)
CREATE NONCLUSTERED INDEX IX_Diccionario_CaracterOriginal ON dbo.Diccionario(CaracterOriginal);
GO

CREATE NONCLUSTERED INDEX IX_Frases_LetrasOcultas ON dbo.Frases(LetrasOcultas);

-- ═══════════════════════════════════════════════════════════════════════════════
-- [08] FLASHCARDS
-- Fichas individuales que rotan 180° mostrando significado y fonética
-- ═══════════════════════════════════════════════════════════════════════════════
PRINT '>>> CREANDO TABLA [08] Flashcards...';
GO

CREATE TABLE Flashcards (
    IdFlashcard     INT IDENTITY(1,1) PRIMARY KEY,
    IdMazo          INT             NOT NULL,

    CONSTRAINT FK_Flashcards_Mazo FOREIGN KEY (IdMazo)

    -- Índices (Recomendación #3)

GO

CREATE NONCLUSTERED INDEX IX_Flashcards_IdMazo_Orden ON dbo.Flashcards(IdMazo, OrdenEnMazo);

-- ═══════════════════════════════════════════════════════════════════════════════
-- CAPA 3: USUARIOS Y CONFIGURACIÓN (4 tablas)
-- Gestión de soldados, preferencias y registro de alistamiento
-- ═══════════════════════════════════════════════════════════════════════════════

-- ═══════════════════════════════════════════════════════════════════════════════
-- [09] USUARIOS (Soldados)
-- Control de vidas, racha diaria y rangos militares
-- ═══════════════════════════════════════════════════════════════════════════════
PRINT '>>> CREANDO TABLA [09] Usuarios...';
GO

CREATE TABLE Usuarios (
    IdUsuario           INT IDENTITY(1,1) PRIMARY KEY,
    IdRango             INT             DEFAULT 1,
    IdIdiomaPreferido   INT             NULL,
    NombreClave         NVARCHAR(100)   NOT NULL,    -- Username único estilo militar
    HashContrasena      NVARCHAR(255)   NOT NULL,    -- BCrypt/Argon2 hash
    VidasActuales       INT             DEFAULT 5,
    RachaDias           INT             DEFAULT 0,
    PuntosTotales       INT             DEFAULT 0,   -- XP acumulado
    UltimaConexion      DATETIME2       DEFAULT GETDATE(),
    FechaRegistro       DATETIME2       DEFAULT GETDATE(),
    EstadoCuenta        NVARCHAR(20)    DEFAULT 'ACTIVA', -- 'ACTIVA', 'SUSPENDIDA', 'BAJA'

    -- Foreign Keys
    CONSTRAINT FK_Usuarios_Rango FOREIGN KEY (IdRango) 
        REFERENCES Rangos(IdRango) ON DELETE SET DEFAULT,
    CONSTRAINT FK_Usuarios_Idioma FOREIGN KEY (IdIdiomaPreferido) 
        REFERENCES Idiomas(IdIdioma) ON DELETE SET NULL,

    -- Constraints
    CONSTRAINT UQ_Usuarios_NombreClave UNIQUE (NombreClave),
    CONSTRAINT CK_Usuarios_Vidas CHECK (VidasActuales >= 0 AND VidasActuales <= 5),
    CONSTRAINT CK_Usuarios_Racha CHECK (RachaDias >= 0),

    -- Índices (Recomendación #3)
);
GO

CREATE NONCLUSTERED INDEX IX_Usuarios_NombreClave ON dbo.Usuarios(NombreClave);
GO

CREATE NONCLUSTERED INDEX IX_Usuarios_IdRango ON dbo.Usuarios(IdRango);
GO

CREATE NONCLUSTERED INDEX IX_Usuarios_IdIdiomaPreferido ON dbo.Usuarios(IdIdiomaPreferido);
GO

CREATE NONCLUSTERED INDEX IX_Usuarios_UltimaConexion ON dbo.Usuarios(UltimaConexion);
GO

    -- Índices

-- [13] SESIONES DE ENTRENAMIENTO
-- Tracking del ciclo de vida del "Soldado" durante cada sesión
-- ═══════════════════════════════════════════════════════════════════════════════
PRINT '>>> CREANDO TABLA [13] SesionesEntrenamiento...';
GO

CREATE TABLE SesionesEntrenamiento (
    IdSesion        INT IDENTITY(1,1) PRIMARY KEY,
    IdUsuario       INT             NOT NULL,
    FechaInicio     DATETIME2       DEFAULT GETDATE(),
    FechaFin        DATETIME2       NULL,
    VidasInicio     INT             DEFAULT 5,
    VidasFinal      INT             NULL,
    PuntajeTotal    INT             DEFAULT 0,
    EstadoSesion    NVARCHAR(20)    DEFAULT 'ACTIVA', -- 'ACTIVA', 'COMPLETADA', 'GAME_OVER', 'ABANDONADA'

    CONSTRAINT CK_Sesiones_VidasInicio CHECK (VidasInicio >= 0 AND VidasInicio <= 5),
    CONSTRAINT CK_Sesiones_Estado CHECK (EstadoSesion IN ('ACTIVA', 'COMPLETADA', 'GAME_OVER', 'ABANDONADA')),

);

-- ═══════════════════════════════════════════════════════════════════════════════
-- [15] HISTORIAL DE ERRORES
-- Algoritmo de repetición espaciada basado en fallos
-- ÍNDICES CRÍTICOS para optimizar la repetición espaciada (Recomendación #3)
-- ═══════════════════════════════════════════════════════════════════════════════
PRINT '>>> CREANDO TABLA [15] HistorialErrores...';
GO

CREATE TABLE HistorialErrores (
    IdError             INT IDENTITY(1,1) PRIMARY KEY,
    IdUsuario           INT             NOT NULL,
    IdItem              INT             NOT NULL,
    FechaFallo          DATETIME2       DEFAULT GETDATE(),
